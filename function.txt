CREATE OR REPLACE FUNCTION FN2(N NUMBER)
RETURN NUMBER 
IS
  S EMPLOYEE.SAL%TYPE;   
BEGIN
  SELECT SAL 
  INTO S
  FROM EMPLOYEE
  WHERE ENO = N;

  RETURN S * 0.10;
END;
/

COMMFUN.SQL
-----------
SET SERVEROUTPUT ON;

DECLARE
  N EMPLOYEE.ENO%TYPE := &N;
  COMM NUMBER;
BEGIN
  COMM := FN2(N);
  DBMS_OUTPUT.PUT_LINE('COMMISSION IS ' || COMM);
END;
/
