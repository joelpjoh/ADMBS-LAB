CREATE OR REPLACE FUNCTION FUN1(A NUMBER, B NUMBER)RETURN NUMBER IS
C NUMBER;
BEGIN C := A+B;
RETURN C;
END;
/

CALLING PROGRAM
---------------
SET SERVEROUTPUT ON;

DECLARE
  X NUMBER := &X;
  Y NUMBER := &Y;
  Z NUMBER;
BEGIN
  Z := FUN1(X, Y);
  DBMS_OUTPUT.PUT_LINE('SUM IS ' || Z);
END;
/
