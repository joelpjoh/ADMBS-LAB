CREATE TABLE EMPLOYEE(ENO NUMBER(4), ENAME VARCHAR(14), DGN VARCHAR(18), SAL NUMBER(6), DEPTNO NUMBER(2));

SQL> INSERT INTO EMPLOYEE VALUES (1010, 'SREE', 'MANAGER', 50000, 12);

SQL> INSERT INTO EMPLOYEE VALUES (1011, 'REEMA', 'ENGINEER', 25000, 13);

SQL> INSERT INTO EMPLOYEE VALUES (1012, 'ANJANA', 'TESTER', 25000, 14);

SET SERVEROUTPUT ON

ED.EMPLOYEE.SQL

DECLARE
N NUMBER(4) := &N;
EN VARCHAR(20);
S NUMBER(6);
BEGIN
SELECT ENO, ENAME, SAL INTO N, EN, S FROM EMPLOYEE WHERE ENO = N;
DBMS_OUTPUT.PUTLINE(N ||" "||EN||" "||S);
END;
